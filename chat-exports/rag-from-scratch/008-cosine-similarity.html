Step 12 — Complete similarity.py

similarity.py
import numpy as np


def cosine_similarity(a: np.ndarray, b: np.ndarray) -> float:
    if a.ndim != 1 or b.ndim != 1 or a.shape != b.shape or a.size == 0:
        raise ValueError("Expected equal-length nonempty vectors")
    if not np.isfinite(a).all() or not np.isfinite(b).all():
        raise ValueError("Vectors must contain finite values")

    a = a.astype(np.float64)
    b = b.astype(np.float64)
    norm_a = np.linalg.norm(a)
    norm_b = np.linalg.norm(b)

    if norm_a == 0 or norm_b == 0:
        raise ValueError("Cosine similarity is undefined for zero vectors")

    dot_product = np.dot(a, b)
    score = dot_product / (norm_a * norm_b)
    return float(np.clip(score, -1.0, 1.0))


def cosine_scores(query: np.ndarray, matrix: np.ndarray) -> np.ndarray:
    if query.ndim != 1 or matrix.ndim != 2:
        raise ValueError("Expected a query vector and an embedding matrix")
    if query.size == 0 or matrix.shape[1] != query.size:
        raise ValueError("Embedding dimensions must match")

    return np.array([cosine_similarity(query, row) for row in matrix])

Step 14 — Complete main.py

main.py
from pathlib import Path
from loader import load_documents
from preprocessor import clean_text
from chunker import create_chunks
from embedder import load_embedding_model, embed_texts, MODEL_NAME
from vector_store import save_vector_store, load_vector_store
from similarity import cosine_scores


def main():
    print("RAG Application Started")

    data_directory = Path(__file__).resolve().parent / "data"
    documents = load_documents(data_directory)
    print(f"Loaded {len(documents)} documents")
    all_chunks = []
    for name, content in documents.items():
        cleaned_content = clean_text(content)
        all_chunks.extend(create_chunks(name, cleaned_content))

    print(f"Created {len(all_chunks)} chunks")
    if not all_chunks:
        print("No chunks to embed.")
        return

    texts = [chunk["text"] for chunk in all_chunks]
    model = load_embedding_model()
    embeddings = embed_texts(model, texts)

    print(f"Embedding matrix shape: {embeddings.shape}")
    for chunk, vector in zip(all_chunks, embeddings, strict=True):
        print(f"{chunk['id']}: {len(vector)} dimensions")

    storage_directory = data_directory.parent / "storage"
    save_vector_store(storage_directory, all_chunks, embeddings, MODEL_NAME)
    print("Vector store saved.")

    saved_chunks, saved_embeddings = load_vector_store(
        storage_directory, MODEL_NAME
    )
    print(f"Reloaded {len(saved_chunks)} chunks")
    print(f"Reloaded matrix shape: {saved_embeddings.shape}")

    reference = saved_embeddings[0]
    scores = cosine_scores(reference, saved_embeddings)
    print(f"\nSimilarity to reference chunk: {saved_chunks[0]['id']}")
    for chunk, score in zip(saved_chunks, scores, strict=True):
        print(f"{chunk['id']}: {score:.4f}")


if __name__ == "__main__":
    main()
