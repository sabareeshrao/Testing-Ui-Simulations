vector_store.py
import json
from pathlib import Path

import numpy as np


def validate_store(chunks: list[dict[str, str]], embeddings: np.ndarray) -> None:
    if embeddings.ndim != 2 or embeddings.shape[1] == 0:
        raise ValueError("Embeddings must be a nonempty-width matrix")
    if len(chunks) != embeddings.shape[0]:
        raise ValueError("Chunk count and vector count must match")
    if not np.isfinite(embeddings).all():
        raise ValueError("Embeddings contain non-finite values")


def save_vector_store(directory, chunks, embeddings, model_name):
    validate_store(chunks, embeddings)
    directory.mkdir(parents=True, exist_ok=True)

    np.save(directory / "embeddings.npy", embeddings, allow_pickle=False)
    metadata = {"model_name": model_name, "chunks": chunks}
    (directory / "chunks.json").write_text(
        json.dumps(metadata, ensure_ascii=False, indent=2),
        encoding="utf-8",
    )


def load_vector_store(directory: Path, model_name: str):
    metadata = json.loads(
        (directory / "chunks.json").read_text(encoding="utf-8")
    )
    if metadata["model_name"] != model_name:
        raise ValueError("Embedding model mismatch")

    chunks = metadata["chunks"]
    embeddings = np.load(directory / "embeddings.npy", allow_pickle=False)
    validate_store(chunks, embeddings)
    return chunks, embeddings
main.py
from pathlib import Path
from loader import load_documents
from preprocessor import clean_text
from chunker import create_chunks
from embedder import load_embedding_model, embed_texts, MODEL_NAME
from vector_store import save_vector_store, load_vector_store


def main():
    print("RAG Application Started")

    data_directory = Path(__file__).resolve().parent / "data"
    documents = load_documents(data_directory)
    print(f"Loaded {len(documents)} documents")
    all_chunks = []
    for name, content in documents.items():
        cleaned_content = clean_text(content)
        all_chunks.extend(create_chunks(name, cleaned_content))

    print(f"Created {len(all_chunks)} chunks")
    if not all_chunks:
        print("No chunks to embed.")
        return

    texts = [chunk["text"] for chunk in all_chunks]
    model = load_embedding_model()
    embeddings = embed_texts(model, texts)

    print(f"Embedding matrix shape: {embeddings.shape}")
    for chunk, vector in zip(all_chunks, embeddings, strict=True):
        print(f"{chunk['id']}: {len(vector)} dimensions")

    storage_directory = data_directory.parent / "storage"
    save_vector_store(storage_directory, all_chunks, embeddings, MODEL_NAME)
    print("Vector store saved.")

    saved_chunks, saved_embeddings = load_vector_store(
        storage_directory, MODEL_NAME
    )
    print(f"Reloaded {len(saved_chunks)} chunks")
    print(f"Reloaded matrix shape: {saved_embeddings.shape}")


if __name__ == "__main__":
    main()
