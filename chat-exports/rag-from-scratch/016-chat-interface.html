<script>
const form = document.getElementById("chat-form");
const question = document.getElementById("question");
const messages = document.getElementById("messages");
const send = document.getElementById("send");
const clear = document.getElementById("clear");
const status = document.getElementById("status");
let busy = false;

function addMessage(role, text) {
    const item = document.createElement("article");
    item.className = "message " + role;
    item.textContent = text;
    messages.append(item);
    return item;
}

function showAnswer(data) {
    const item = addMessage("assistant", data.answer);
    for (const source of data.sources) {
        const detail = document.createElement("div");
        detail.className = "source";
        detail.textContent = source.id + " | " + source.source + "\n" + source.text;
        item.append(detail);
    }
    const note = document.createElement("p");
    note.className = "warning";
    note.textContent = "Citation status: " + data.citation_status;
    if (data.unknown_citations.length) {
        note.textContent += " | Unknown IDs: " + data.unknown_citations.join(", ");
    }
    item.append(note);
}

form.addEventListener("submit", async (event) => {
    event.preventDefault();
    const text = question.value.trim();
    if (busy || !text || text.length > 2000) return;
    busy = true;
    send.disabled = clear.disabled = question.disabled = true;
    addMessage("user", text);
    question.value = "";
    status.textContent = "Retrieving and generating...";
    try {
        const response = await fetch("/ask", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ question: text })
        });
        if (!response.ok) throw new Error("HTTP " + response.status);
        showAnswer(await response.json());
        status.textContent = "Ready";
    } catch (error) {
        addMessage("assistant", "Unable to answer. Please try again.");
        status.textContent = "Request failed";
    } finally {
        busy = false;
        send.disabled = clear.disabled = question.disabled = false;
        question.focus();
    }
});

clear.addEventListener("click", () => {
    if (busy) return;
    messages.replaceChildren();
    question.value = "";
    status.textContent = "Ready";
    question.focus();
});

Step 4 — Complete api.py

api.py
from contextlib import asynccontextmanager
from pathlib import Path
from typing import Annotated, Literal

from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import FileResponse
from openai import OpenAIError
from pydantic import BaseModel, StringConstraints

from rag_service import RAGService


class AskRequest(BaseModel):
    question: Annotated[str, StringConstraints(
        strip_whitespace=True, min_length=1, max_length=2000
    )]


class Source(BaseModel):
    id: str
    source: str
    text: str


class AskResponse(BaseModel):
    status: Literal["answered", "no_context"]
    answer: str
    sources: list[Source]
    unknown_citations: list[str]
    citation_status: Literal[
        "id_match_only", "unknown_ids", "no_citations", "not_applicable"
    ]


@asynccontextmanager
async def lifespan(app: FastAPI):
    data_directory = Path(__file__).resolve().parent / "data"
    app.state.rag = RAGService(data_directory)
    yield


app = FastAPI(title="RAG from Scratch", lifespan=lifespan)


@app.get("/", include_in_schema=False)
def chat_page():
    return FileResponse(Path(__file__).resolve().parent / "chat.html")


@app.get("/health")
def health():
    return {"status": "ready"}


@app.post("/ask", response_model=AskResponse)
def ask(payload: AskRequest, request: Request) -> AskResponse:
    try:
        result = request.app.state.rag.ask(payload.question)
        return AskResponse(**result)
    except (OpenAIError, ValueError):
        raise HTTPException(status_code=502, detail="Answer unavailable.") from None
